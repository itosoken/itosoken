<iframe id="koshihaken" src="https://itosoken.github.io/itosoken/koshihaken/" style="width:100%;border:0;display:block;" height="3000" title="ケアマネジャー試験対策講座 講師派遣のご案内"></iframe>
<script>
window.addEventListener("message", function(e){
  if (e.origin !== "https://itosoken.github.io") return;
  var f = document.getElementById("koshihaken");
  if (!e.data || !f) return;
  if (e.data.type === "koshihaken-height") {
    f.style.height = e.data.height + "px";
  }
  if (e.data.type === "koshihaken-contact") {
    var y = f.getBoundingClientRect().bottom + window.pageYOffset - 90;
    window.scrollTo({ top: y, behavior: "smooth" });
  }
});
</script>

window.addEventListener("message", function(e){
  if (e.origin !== "https://itosoken.github.io") return;
  var f = document.getElementById("kyotsu-guide");
  if (!e.data || !f) return;
  if (e.data.type === "kyotsu-guide-height") {
    f.style.height = e.data.height + "px";
  }
  if (e.data.type === "kyotsu-guide-scroll") {
    var y = f.getBoundingClientRect().top + window.pageYOffset + e.data.top - 100;
    window.scrollTo({ top: y, behavior: "smooth" });
  }
});
</script>
